from pathlib import Path

 for fold in range(38):
  ps=[]
  for seed in [42,43,44]:
   n=next(n for n in z.namelist() if n.startswith('runs/'+c+'_') and f'/seed_{seed}/fold_{fold:03d}/validation_predictions.parquet' in n)
   p=pd.read_parquet(io.BytesIO(z.read(n)));ps.append(p)

   chosen=h[h.epoch==m['best_epoch']].iloc[0];oracle=h.loc[h.average_precision.idxmax()]
   history.append({'candidate':c,'fold':fold,'seed':seed,'chosen_epoch':m['best_epoch'],'chosen_bce':chosen.bce,'chosen_ap':chosen.average_precision,'max_ap_epoch_diagnostic':oracle.epoch,'max_ap_diagnostic':oracle.average_precision,'bce_at_max_ap':oracle.bce,'last_bce':h.bce.iloc[-1],'last_train_bce':h.train_bce.iloc[-1],'chosen_train_bce':chosen.train_bce})

     if slope<=0:a=np.full(len(fit),prior);b=np.full(len(ev),prior);fallback=True
     else:a=model.predict_proba(fit.reshape(-1,1))[:,1];b=model.predict_proba(ev.reshape(-1,1))[:,1]
    else:
     model=IsotonicRegression(out_of_bounds='clip',y_min=0,y_max=1).fit(fit,fy);a=model.predict(fit);b=model.predict(ev)
    assert len(b)==507 and np.isfinite(b).all() and ((b>=0)&(b<=1)).all()
    bce=log_loss(ey,b,labels=[0,1]);ref=log_loss(ey,np.full(len(ey),prior),labels=[0,1]);brier=brier_score_loss(ey,b)
    row={'candidate':c,'fold':fold,'channel':channel,'method':method,'fit_end':str(ts.iloc[499]),'assessment_start':str(ts.iloc[510]),'n':len(ey),'fit_prevalence':prior,'assessment_prevalence':ey.mean(),'ap':average_precision_score(ey,b),'auc':roc_auc_score(ey,b),'bce':bce,'bce_reference':ref,'bce_skill':1-bce/ref,'brier':brier,'slope':slope,'fallback':fallback,'max_score':max(b)}
    for label,t in [('t050',.5),('t070',.7),('early_q90',float(np.quantile(a,.9)))]:
     selected=b>=t;tp=int(ey[selected].sum());nsel=int(selected.sum());row.update({label+'_threshold':t,label+'_signals':nsel,label+'_tp':tp,label+'_precision':tp/nsel if nsel else np.nan,label+'_recall':tp/ey.sum(),label+'_rate':nsel/len(ey)})
    rows.append(row)
