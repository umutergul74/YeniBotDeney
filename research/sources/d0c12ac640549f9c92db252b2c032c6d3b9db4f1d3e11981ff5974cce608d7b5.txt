from pathlib import Path
import json, hashlib
import numpy as np
import pandas as pd
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
from sklearn.metrics import average_precision_score

ROOT = Path(__file__).resolve().parents[2]
OUT = ROOT/'research/analyses/question-recap-20261017'
DETAIL = Path(__file__).parent
LEGACY = ROOT/'evidence/drive_snapshot/colab_v1'
manifest = ROOT/'evidence/colab_development/review_manifest.json'
checksums = {r['path']: r['sha256'] for r in json.loads(manifest.read_text())['files']}
inputs = []
def checked(p, expected):
    digest = hashlib.sha256(p.read_bytes()).hexdigest()
    assert digest == expected, str(p)
    inputs.append(dict(path=p.relative_to(ROOT).as_posix(), sha256=digest))
    return p
def archived(p):
    return checked(p, checksums[p.relative_to(LEGACY).as_posix()])

grid = np.arange(201)/200
all_curves, scores, anchors = [], [], {}
names = {'A':'A · 6 özellik · GRU', 'B':'B · 34 özellik · GRU',
         'C':'C · 34 özellik · TCN', 'D':'D · 34 özellik · Paralel TCN+GRU'}
for model in names:
    runs = list((LEGACY/'runs').glob(model+'_*'))
    assert len(runs)==1
    run=runs[0]
    protocol=json.loads(archived(run/'protocol.json').read_text())
    assert protocol['test_evaluations']==0
    baseline=pd.read_csv(archived(run/'validation_summary.csv')).set_index(['seed','fold'])
    assert baseline.index.is_unique and set(baseline.index)=={(s,f) for s in [42,43,44] for f in range(38)}
    for seed in [42,43,44]:
        for fold in range(38):
            p=pd.read_parquet(archived(run/f'seed_{seed}/fold_{fold:03d}/validation_predictions.parquet'))
            p.timestamp=pd.to_datetime(p.timestamp, utc=True)
            assert len(p)==1017 and p.timestamp.is_unique and p.timestamp.is_monotonic_increasing
            assert p.timestamp.max()<pd.Timestamp('2026-01-01',tz='UTC')
            identity=p[['timestamp','source_row_position','label','forward_return']]
            if model=='A': anchors[(seed,fold)]=identity
            else: pd.testing.assert_frame_equal(identity, anchors[(seed,fold)])
            y=p.label.to_numpy().astype(int); prob=p.probability.to_numpy().astype(float)
            assert np.isin(y,[0,1]).all() and np.isfinite(prob).all() and ((prob>=0)&(prob<=1)).all()
            decisions=prob[:,None]>=grid[None,:]
            n=decisions.sum(0); tp=(decisions & (y[:,None]==1)).sum(0)
            fp=n-tp; fn=y.sum()-tp
            precision=np.divide(tp,n,out=np.zeros(201),where=n>0)
            recall=tp/y.sum()
            f1=2*tp/(2*tp+fp+fn)
            for k,v in [('precision',precision),('recall',recall),('f1',f1)]:
                assert np.isclose(v[100],baseline.loc[(seed,fold),k],atol=1e-8), (model,seed,fold,k)
            c=pd.DataFrame(dict(model=model,seed=seed,fold=fold,threshold=grid,
                precision=precision,recall=recall,f1=f1,signals=n,signal_rate=n/len(y),
                precision_active=np.where(n>0,precision,np.nan),no_signal=(n==0).astype(int)))
            all_curves.append(c)
            q=np.clip(prob,1e-7,1-1e-7)
            scores.append(dict(model=model,seed=seed,fold=fold,prevalence=y.mean(),
                ap=average_precision_score(y,prob),bce=-(y*np.log(q)+(1-y)*np.log(1-q)).mean(),
                positive_mean_score=prob[y==1].mean(),negative_mean_score=prob[y==0].mean()))
    print(model, '114 validation groups verified', flush=True)
curves=pd.concat(all_curves,ignore_index=True)
curves.to_csv(DETAIL/'thresholds_by_fold_seed.csv',index=False)
summary=curves.groupby(['model','threshold']).agg(precision=('precision','mean'),recall=('recall','mean'),
    f1=('f1','mean'),signal_rate=('signal_rate','mean'),mean_signals=('signals','mean'),
    precision_active=('precision_active','mean'),no_signal_groups=('no_signal','sum')).reset_index()
summary.to_csv(OUT/'abcd_thresholds.csv',index=False)
summary[summary.threshold.isin([.2,.3,.4,.5,.6,.7])].to_csv(OUT/'abcd_selected_thresholds.csv',index=False)
curves.groupby(['model','seed','threshold'])[['precision','recall','f1','signal_rate']].mean().to_csv(OUT/'abcd_seed_curves.csv')
metrics=pd.DataFrame(scores).groupby('model').mean(numeric_only=True).drop(columns=['seed','fold'])
metrics.to_csv(OUT/'abcd_score_metrics.csv')
old=pd.read_csv(ROOT/'research/analyses/diagnostics-20261015/results/threshold_summary.csv')
for r in old.itertuples():
    a=summary[(summary.model=='A') & np.isclose(summary.threshold,r.threshold)].iloc[0]
    for key in ['precision','recall','f1']: assert np.isclose(a[key],getattr(r,key),atol=1e-10)

plt.rcParams.update({'font.family':'DejaVu Sans','font.size':10,'axes.spines.top':False,'axes.spines.right':False})
fig,axes=plt.subplots(2,2,figsize=(13,9),layout='constrained')
for ax,(model,name) in zip(axes.flat,names.items()):
    s=summary[summary.model==model]
    for col,label,color in [('precision','Precision','#2563eb'),('recall','Recall','#0d9488'),('f1','F1','#d97706'),('signal_rate','Sinyal oranı','#64748b')]:
        ax.plot(s.threshold,s[col]*100,label=label,color=color,lw=2)
    ax.axvline(.5,color='black',ls=':',lw=1)
    ax.set(title=name,xlabel='Karar eşiği',ylabel='%',xlim=(0,1),ylim=(0,102))
    ax.grid(alpha=.15)
axes[0,0].legend(loc='upper right',fontsize=9)
fig.suptitle('Özgün A/B/C/D · validation eşik–performans eğrileri\n38 fold × 3 seed eşit ortalama; 2/5 ATR; 10 saat ufuk',fontsize=15)
fig.supxlabel('Sinyalsiz gruplarda precision=0 (eski rapor kuralı). Eşik seçilmedi; test kullanılmadı.',fontsize=10)
fig.savefig(OUT/'abcd_threshold_curves.png',dpi=170); plt.close(fig)

barroot=ROOT/'research/analyses/barrier-expansion-20261015/results'
barhash={r['path']:r['sha256'] for r in json.loads((barroot/'output_manifest.json').read_text())}
def barfile(p):return checked(p,barhash[p.relative_to(ROOT).as_posix()])
table=pd.read_csv(barfile(barroot/'barrier_comparison.csv')).rename(columns={'scenario':'barrier'})
order=['2/5','1/1','1/2','1.5/2','2/2','3/2']
table=table.set_index('barrier').loc[order].reset_index() if 'barrier' in table else table
timeout=[]; samples={}; timestamp_anchor=None
for label in order:
    tp,sl=label.split('/')
    data=pd.read_csv(barfile(ROOT/f'output/advisor_experiments/barrier-expansion-20261015/labels_tp{tp}_sl{sl}.csv'))
    assert len(data)==33551 and data.timestamp.is_unique
    if timestamp_anchor is None: timestamp_anchor=data.timestamp
    else: pd.testing.assert_series_equal(data.timestamp,timestamp_anchor)
    assert np.array_equal(data.label.to_numpy(),(data.hit_code==1).astype(int).to_numpy())
    r=table[table.barrier==label].iloc[0]
    assert (data.hit_code==1).sum()==r.profit_first
    assert data.hit_code.isin([2,3]).sum()==r.stop_first
    assert (data.hit_code==0).sum()==r.timeout
    x=data.loc[data.hit_code==0,'forward_return_10h'].to_numpy()*100
    assert np.isfinite(x).all()
    assert np.isclose(x.mean(),r.timeout_mean_fwd_pct)
    samples[label]=x
    timeout.append(dict(barrier=label,n=len(x),mean_pct=x.mean(),median_pct=np.median(x),
        p05_pct=np.quantile(x,.05),p25_pct=np.quantile(x,.25),p75_pct=np.quantile(x,.75),p95_pct=np.quantile(x,.95),
        positive_pct=100*(x>0).mean(),min_pct=x.min(),max_pct=x.max()))
table.to_csv(OUT/'six_barrier_counts.csv',index=False)
pd.DataFrame(timeout).to_csv(OUT/'six_barrier_timeout_returns.csv',index=False)
fig,ax=plt.subplots(figsize=(11,5.8),layout='constrained')
left=np.zeros(6)
for col,text,color in [('profit_first_pct','TP önce','#0d9488'),('stop_first_pct','SL önce (çift temas dahil)','#dc6666'),('timeout_pct','Süre dolumu','#94a3b8')]:
    vals=table[col].to_numpy(); ax.barh(order,vals,left=left,label=text,color=color)
    for i,v in enumerate(vals): ax.text(left[i]+v/2,i,f'%{v:.1f}',ha='center',va='center',fontsize=10)
    left+=vals
ax.invert_yaxis();ax.set(xlim=(0,100),xlabel='Aynı 33.551 saat içindeki pay (%)',ylabel='TP / SL (ATR)')
ax.set_title('Altı bariyerin etiket dağılımı · 10 saat ufuk',fontsize=15)
ax.legend(loc='upper center',bbox_to_anchor=(.5,-.12),ncol=3)
fig.savefig(OUT/'six_barrier_distribution.png',dpi=170);plt.close(fig)
fig,axes=plt.subplots(2,3,figsize=(13,7.8),layout='constrained')
for ax,label in zip(axes.flat,order):
    x=np.sort(samples[label]); ax.plot(x,np.arange(1,len(x)+1)/len(x)*100,color='#2563eb',lw=2)
    ax.axvline(0,color='black',ls=':',lw=1)
    ax.set(title=f'{label} · n={len(x):,}',xlabel='10 saatlik brüt getiri (%)',ylabel='Birikimli pay (%)',ylim=(0,100))
    ax.grid(alpha=.15)
fig.suptitle('Süre dolan örneklerin getiri dağılımları\nHer panel kendi timeout altkümesini gösterir; strateji getirisi değildir.',fontsize=14)
fig.savefig(OUT/'six_barrier_timeout_ecdf.png',dpi=170);plt.close(fig)
(DETAIL/'input_manifest.json').write_text(json.dumps(inputs,indent=2),encoding='utf-8')
print(summary[summary.threshold==.5].to_string(index=False))
print('Completed: 456 prediction files, 804 threshold summaries, 6 matched barrier tables.')
