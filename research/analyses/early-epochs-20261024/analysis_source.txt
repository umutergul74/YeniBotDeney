import io, json, hashlib, zipfile, re, subprocess
from pathlib import Path
import numpy as np
import pandas as pd
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

ROOT = Path(__file__).resolve().parents[2]
OUT = ROOT/'research/analyses/early-epochs-20261024'
ZIP = Path(__file__).with_name('source.zip')
OUT.mkdir(parents=True, exist_ok=True)
sha = lambda b: hashlib.sha256(b).hexdigest()
z = zipfile.ZipFile(ZIP)
assert z.testzip() is None
protocols = {json.loads(z.read(n))['experiment']: json.loads(z.read(n)) for n in z.namelist() if n.endswith('/protocol.json')}
assert set(protocols) == set('ABCD')
assert len({p['data_sha256'] for p in protocols.values()}) == 1
assert all(p['config'] == protocols['A']['config'] for p in protocols.values())
assert set(protocols['A']['features']) <= set(protocols['B']['features'])
assert protocols['B']['features'] == protocols['C']['features'] == protocols['D']['features']
assert [len(protocols[c]['features']) for c in 'ABCD'] == [6,37,37,37]
source_checks = {}
for path, expected in protocols['A']['source_hashes'].items():
    relative = path.split('/advisor_feature_comparison_repo/')[1]
    raw = subprocess.check_output(['git','show',f'advisor-feature-comparison-v1.0.0:{relative}'],cwd=ROOT)
    source_checks[relative] = sha(raw) == expected
assert all(source_checks.values()), source_checks
rows, histories, hashes = [], [], {}
for name in sorted(n for n in z.namelist() if n.endswith('/history.csv')):
    model, seed, fold = re.match(r'runs/([A-D])_[^/]+/seed_(\d+)/fold_(\d+)/history.csv',name).groups()
    seed,fold=int(seed),int(fold)
    raw=z.read(name); hashes[name]=sha(raw)
    h=pd.read_csv(io.BytesIO(raw))
    m=json.loads(z.read(name.replace('history.csv','validation_metrics.json')))
    assert h.epoch.tolist()==list(range(1,len(h)+1))
    assert np.isfinite(h[['train_bce','bce','average_precision']]).all().all()
    assert (h.samples==1017).all() and h.prevalence.nunique()==1 and (h.threshold==.5).all()
    best=h.loc[h.bce.idxmin()]; first=h.iloc[0]; last=h.iloc[-1]
    assert m['status']=='complete' and m['test_evaluations']==0
    assert m['best_epoch']==int(best.epoch) and m['epochs_trained']==len(h)
    assert len(h)==min(100,int(best.epoch)+15)
    assert m['features']==protocols[model]['features']
    for k in m['validation']:
        if m['validation'][k] is not None:
            assert np.isclose(best[k],m['validation'][k],rtol=1e-7,atol=1e-8),(name,k)
    assert all(a['passed'] for a in m['boundary_audit'].values())
    r=dict(model=model,seed=seed,fold=fold,best_epoch=int(best.epoch),epochs=len(h),parameters=m['parameters'],train_sequences=m['train_sequences'],prevalence=first.prevalence,
           validation_section_start=m['boundary_audit']['train_validation']['next_section_start'])
    for stage,record in [('first',first),('best',best),('last',last),('epoch16',h.iloc[15])]:
        for k in ['train_bce','bce','average_precision','precision','recall','f1','rank_ic']:
            r[f'{stage}_{k}']=record[k]
    r['max_ap_epoch']=int(h.loc[h.average_precision.idxmax(),'epoch'])
    r['max_ap']=h.average_precision.max()
    rows.append(r)
    h=h.assign(model=model,seed=seed,fold=fold)
    histories.append(h)
df=pd.DataFrame(rows); allh=pd.concat(histories,ignore_index=True)
expected={(m,s,f) for m in 'ABCD' for s in (42,43,44) for f in range(38)}
assert set(zip(df.model,df.seed,df.fold))==expected and len(df)==456
assert df.epochs.min()==16
old=pd.read_csv(ROOT/'research/analyses/feature-comparison-review-20261024/metrics_by_fold_seed.csv')
print('prior columns',old.columns.tolist())
paired=df.merge(old,on=['model','seed','fold'],validate='one_to_one',suffixes=('','_old'))
assert len(paired)==456
for ours, prior in [('best_epoch','best_epoch_old'),('best_bce','bce'),('best_average_precision','average_precision')]:
    assert np.allclose(paired[ours],paired[prior],atol=1e-8,rtol=1e-7)
summary=[]
for model,g in df.groupby('model'):
    r={'model':model,'runs':len(g),'parameters':int(g.parameters.iloc[0]),'best_epoch_mean':g.best_epoch.mean(),'best_epoch_median':g.best_epoch.median(),'best_epoch_max':g.best_epoch.max(),
       'best_epoch_1':int((g.best_epoch==1).sum()),'best_epoch_le3':int((g.best_epoch<=3).sum()),'epochs_min':int(g.epochs.min()),'epochs_max':int(g.epochs.max()),
       'train_down_val_up_first_to16':int(((g.epoch16_train_bce<g.first_train_bce)&(g.epoch16_bce>g.first_bce)).sum()),
       'ap_down_first_to16':int((g.epoch16_average_precision<g.first_average_precision).sum()),
       'ap_down_best_to_last':int((g.last_average_precision<g.best_average_precision).sum()),
       'ap_peak_differs_from_bce_min':int((g.max_ap_epoch!=g.best_epoch).sum())}
    for stage in ['first','best','last','epoch16']:
        for k in ['train_bce','bce','average_precision']:
            r[f'{stage}_{k}']=g[f'{stage}_{k}'].mean()
    summary.append(r)
summary=pd.DataFrame(summary)
df.to_csv(OUT/'run_diagnostics.csv',index=False)
summary.to_csv(OUT/'model_summary.csv',index=False)
allh.to_csv(OUT/'epoch_metrics.csv',index=False)
